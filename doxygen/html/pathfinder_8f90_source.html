46!!---------------------------
47!!
48!! Given input xyz files for target reactants and products, this routine will
49!! try to find a reaction mechanism connecting the two by using graph-moves defined
50!! in the "movefile".
51!!
52!! After determination of a path which definitively connects reactants and products,
53!! we then generate xyz-files for each reaction-path along the reaction-mechanism
54!! using the idea of optimization under the graph-restraining potential.
55!!
56!! If optaftermove = .TRUE. in the input file, this code will also perform geometry
57!! optimization for all of the intermediate structures and report the results in the
58!! .log file.
59!!
60!!--------------------------
61!! OUTPUTS
62!!--------------------------
63!!
64!! After completion, the following outputs are worth looking at:
65!! - mcopt.dat: contains the graph-error function as a function of number of MC iterations
66!! in the calculation. This should reach zero for a successful calculation.
67!!
68!! - final_path.xyz: Contains the intermediate structures (xyz format) along the located reaction
69!! mechanism. If optaftermove = .true., the structures are geometry-optimized using
70!! whichever electronic structure methods is specified in the input file.
71!!
72!! - final_path_rx_*.xyz: This series of xyz files, one for each reaction-step in the mechanism,
73!! contain initial nimage-long paths for each intermediate reaction.
74!!
75!! - .log: The .log file will contain useful information about the reaction, including the
76!! calculated Q-value and the maximum dE value.
77!!
78!! In addition, if igfunc = 4 (so we are targetting formation of a single molecule or
79!! subset of molecules, rather than an entire structure), the further outputs are
80!! produced which have the prefix 'adjusted_'. This are files which have been determined
81!! for the reaction path after removing molecules and reactions which are not relevant
82!! to the formation of the product.

411 ! endif

481 !************************************************************************
482 !> AdjustPaths
483 !!
484 !! This routine performs a series of functions to
485 !! simplify the post-analysis of reaction-path simulations.
486 !! (i) If pesfull = .FALSE., we print out the energies and Q-values
487 !! calculated only for those molecules involved in formation of the
488 !! product.
489 !! (ii) We print out a new set of reaction paths which only involve
490 !! the molecules containing atoms ending up in the final structure.
491 !! (iii) We strip the adjusted paths of non-reactive molecules.
492 !!
493 !! The final set of xyz path files should then be most appropriate
494 !! for post-analysis.
495 !!
496 !! Input parameters:
497 !! - cx_start: CXS object for reactants.
498 !! - cx_end: CXS object for target product.
499 !! - cx(nrxn): Intermediate CXS objects along mechanism path.
500 !! - nrxn: Maximum number of active reactions.
501 !! - movenum(:): Graph-move number at each reaction step.
502 !! - moveatoms(:,:): Atoms involved in graph-moves at each step.
503 !! - ifound: Reaction step-number at which target products are first
504 !! found.
505 !!
506 !************************************************************************
507 Subroutine adjustpaths(cx_start, cx_end, cx, nrxn, movenum, moveatoms, ifound )
508 implicit none
509 integer :: i,j,k,nrxn, ifound,itarg,irxn,ncount,vtot_last,iend,jj
510 type(cxs) :: cx_start, cx_end, cx(nrxn),cx1,cx2
511 integer :: movenum(nrxn), moveatoms(nrxn,NAMOVEMAX),im,imove
512 integer :: m, ii, n1
513 real(8) :: grapherror,Qvalue,Qvalue2,dE,vtot,vtot_start,Qmax,v1,v2
514 real(8) :: drsq1, drsq2, xtemp, ytemp, ztemp,dx,dy,dz,de_sum
515 logical :: errflag, active(0:NRXNMAX,NMOLMAX),barrierless
516 character (len=2) :: ctemp
517 character (len=7) :: x1
518 character (len=8) :: fmt
519 character (len=20) :: file_root
520
521 write(logfile,'(/"================================================================"/)')
522 write(logfile,'(/"* Adjusting paths to focus on target formation..."/)')
523
524 ! Propagate the graph from the start to generate the final graph:
525 !
526 ! Call PropagateGraphs( cx_start, cx, nrxn, movenum, moveatoms, errflag, GraphError )
527
528 ! Determine which reaction-step has first appearance of products.
529 !
530 write(logfile,'("* Target found in reaction:",2x,i4)')ifound
531 write(logfile,'("* Molecule ID of target is:",2x,i4)')cx(ifound)%itargetmol
532 call flush(logfile)
533 itarg = cx(ifound)%itargetmol
534
535
536 ! We now need to step backwards from the frame in which the target molecules
537 ! were found and keep track of all the molecules containing atoms..
538 !
539 ! First, set the active flag to true for the product molecule (itarg) in the frame
540 ! when it is found (ifound).
541 !
542 active(:,:) = .false.
543 active(ifound,itarg) = .true.
544
545 ! Now for the ifound-1 frame, work out the active molecules - this is neccesary,
546 ! so when can then recheck the ifound frame to account for active molecules
547 ! formed by e.g. dissociation, which we might have missed if we only set the
548 ! target molecule flag to "active".
549 !
550 do jj = 1, nrxn
551
552 irxn = ifound - 1
553 do i = 1, cx(irxn)%nmol
554 do j = 1, cx(irxn+1)%nmol
555 if (active(irxn+1,j)) then
556 if (containstargetatom(cx(irxn),i,cx(irxn+1),j)) then
557 active(irxn,i) = .true.
558 endif
559 endif
560 enddo
561 enddo
562
563 ! Now, based on the active molecules in ifound-1, make sure that everything
564 ! is activated in frame ifound.
565 !
566 do i = 1, cx(ifound)%nmol
567 do j = 1, cx(ifound-1)%nmol
568 if (active(ifound-1,j)) then
569 if (containstargetatom(cx(ifound),i,cx(ifound-1),j)) then
570 active(ifound,i) = .true.
571 endif
572 endif
573 enddo
574 enddo
575
576
577 ! Now that we've identified molecules which are active in the ifound
578 ! frame, go back and identify active molecules in other frames.
579 !
580 do irxn = ifound-1, 0, -1
581
582 ! Run through the molecules in this frame and check whether they
583 ! contain any of the active molecules in the previous frame.
584 !
585 if (irxn > 0 ) then
586
587 do i = 1, cx(irxn)%nmol
588 do j = 1, cx(irxn+1)%nmol
589 if (active(irxn+1,j)) then
590 if (containstargetatom(cx(irxn),i,cx(irxn+1),j)) then
591 active(irxn,i) = .true.
592 endif
593 endif
594 enddo
595 enddo
596
597 else if (irxn == 0) then
598
599 do i = 1, cx_start%nmol
600 do j = 1, cx(irxn+1)%nmol
601 if (active(irxn+1,j))then
602 if (containstargetatom(cx_start,i,cx(irxn+1),j)) then
603 active(irxn,i) = .true.
604 endif
605 endif
606 enddo
607 enddo
608
609 endif
610 enddo
611
612 ! Now go forwards, activating molecules.
613 !
614 do irxn = 1, ifound
615
616 ! Run through the molecules in this frame and check whether they
617 ! contain any of the active molecules in the previous frame.
618 !
619 if (irxn == 1 ) then
620
621 do i = 1, cx(irxn)%nmol
622 do j = 1, cx_start%nmol
623 if (active(irxn-1,j)) then
624 if (containstargetatom(cx(irxn),i,cx_start,j)) then
625 active(irxn,i) = .true.
626 endif

942 ! reactant structure...

945 do j = 1, cx2%na ! Loop over atoms in cx1
946
947 if (cx1%atomlabel(k) == cx2%atomlabel(j)) then
948
949 dx = cx2%r(1,j) - cx1%r(1,k)
950 dy = cx2%r(2,j) - cx1%r(2,k)
951 dz = cx2%r(3,j) - cx1%r(3,k)
952 drsq1 = dx*dx + dy*dy + dz*dz
953 dx = cx2%r(1,j) - cx1%r(1,j)
954 dy = cx2%r(2,j) - cx1%r(2,j)
955 dz = cx2%r(3,j) - cx1%r(3,j)
956 drsq2 = dx*dx + dy*dy + dz*dz
957
958 ! if drsq1 < drsq2, j is closer to k, so swap!
959 if ( drsq1 < drsq2 ) then
960 ctemp = cx2%atomlabel(j)
961 xtemp = cx2%r(1,j)
962 ytemp = cx2%r(2,j)
963 ztemp = cx2%r(3,j)
964 cx2%atomlabel(j) = cx2%atomlabel(k)
965 cx2%r(1:3,j) = cx2%r(1:3,k)
966 cx2%atomlabel(k) = ctemp
967 cx2%r(1,k) = xtemp
968 cx2%r(2,k) = ytemp
969 cx2%r(3,k) = ztemp
970 endif
971 endif
972 enddo
973 enddo
974 enddo
975
976 ! Finally, print the two reordered structures to file.
977 !
978 Call printcxstofile(cx1,'reorder_reactants_'//trim(x1)//'.xyz',0.d0)
979 Call printcxstofile(cx2,'reorder_products_'//trim(x1)//'.xyz',0.d0)
980
981 CAll deletecxs(cx1)
982 Call deletecxs(cx2)
983
984 enddo
985
986 return
987 end Subroutine adjustpaths

1064 !!
1065 !! Compare two separate structures via their graphs. In particular,
1066 !! we compare the eigenvalues of the atomic-weighted graph Coulomb
1067 !! matrix, with the distances between atoms given by the shortest-path
1068 !! distances as calculated from the graph.
1069 !!
1070 !! cx1, cx2 - Two structures to compare. Note that the graphs MUST have
1071 !! been already calculated.
1072 !! error - Calculate mean-square error between eigenvalues.
1073 !!
1074 !************************************************************************
1075 !
1076 Subroutine comparegraphs(cx1, cx2, error)
1077 implicit none

1079 integer :: i1,j1,ina
1080 type(cxs) :: cx1, cx2

1088 integer, allocatable :: nelid(:)
1089 character*2 :: label

1091 na = cx1%na
1092 error = 0.d0
1093
1094
1095 ! IGFUNC = 0: Use element-wise comparison of graphs (NOT PERMUTATIONALLY INVARIANT, AFFECTED
1096 ! BY ORDER OF ATOMS IN INPUT FILES):
1097 !

1099
1100 do i = 1, na-1
1101 do j = i+1,na
1102 error = error + (cx1%graph(i,j)-cx2%graph(i,j))**2

1104 enddo

1106 else if (igfunc == 1 .or. igfunc == 3) then
1107
1108 ! allocate( dist1(na,na), dsp1(na,na) )

1110 ! do i = 1, na
1111 ! do j = i, na

1114 ! dist1(j,i) = 1.d0
1115 ! else
1116 ! dist1(i,j) = BIG
1117 ! dist1(j,i) = BIG
1118 ! endif
1119 ! enddo
1120 ! enddo
1121 ! Call GetShortestPaths(na,dist1,dsp1)
1122 !

1124 ! do i = 1, na
1125 ! do j = i, na
1126 ! if ( cx2%graph(i,j) == 1) then
1127 ! dist2(i,j) = 1.d0
1128 ! dist2(j,i) = 1.d0
1129 ! else

1133 ! enddo
1134 ! enddo
1135 ! Call GetShortestPaths(na,dist2,dsp2)

1138 !

1164 a2(i,i) = id1

1187 ! deallocate(dsp2)
1188 ! deallocate(dist2)

1278 ! of molecules.

1283 !
1284 Call getmols(cx1)

1287 ! (which is just a single molecule).
1288 !
