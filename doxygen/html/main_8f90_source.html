43 ! Switch based on calctype.
44 !
45 select case (calctype)
46
47 !***************************************************
48 ! PATH OPTIMIZATION
49 !***************************************************
50 !
51 case('optpath')
52
53 write(logfile,'("* Starting path optimization..."/)')
54
55 ! Initialize the reaction path object.
56 !
57 Call newpath(rp, startfrompath, startfile, endfile, pathfile, nimage, &

59
60 ! Print the initial path.
61 !
62 Call printpathtofile(rp,'initial_path.xyz')
63 write(logfile,'("- Initial path written to initial_path.xyz")')
64
65 ! If stripinactive is true, then we strip out the inactive molecules (i.e. those not
66 ! involved in the reaction).
67 !
68 if (stripinactive) then
69 write(logfile,'("*** Stripping inactive molecules from path...")')
70 Call stripinactivefrompath( rp, 'strip_path.xyz', fixeddof, fixedatom, ndofconstr, natomconstr,.true.)
71 write(logfile,'("- Stripped initial path written to strip_path.xyz")')
72
73 ! Now delete the original path and read the new one.
74 !
75 Call deletepath(rp)
76
77 ! ..and create a new path by reading from file...
78 !
79 Call newpath(rp, .true., startfile, endfile, 'strip_path.xyz', nimage, &
80 pathinit, .false., idum)
81
82 endif
83
84
85 ! Initialize constraints in the path.
86 !
87 Call setpathconstraints(rp, ndofconstr, fixeddof, natomconstr, fixedatom)
88
89 ! Now optimize the path.
90 !
91 Call optimizepath(rp)
92
93 ! Tidy up....
94 Call deletepath(rp)
95
96
97 !***************************************************
98 ! PATH INTERPOLATION
99 !***************************************************
100 !
101 case ('interp')
102
103 write(logfile,'("* Starting path interpolation..."/)')
104
105 ! Create interpolation.
106 call interpolatepath(rp)
107
108 ! Tidy up.
109 call deletepath(rp)
110
111
112 !***************************************************
113 ! DOUBLE-ENDED PATH-FINDING CALCULATION
114 !***************************************************
115 !
116 case ('pathfind')
117
118 write(logfile,'("* Starting double-ended mechanism-finding calculation..."/)')
119 Call runpathfinder()
120
121 !***************************************************
122 ! SINGLE-ENDED NETWORK GENERATION CALCULATION
123 !***************************************************
124 !
125 case ('netgrow')
126
127 write(logfile,'("* Starting single-ended network generation calculation..."/)')
128 write(logfile, '("- Note that this mode is EXPERIMENTAL.")')
129 Call runnetgrow()
130
131 case ('netgrow2')
132
133 write(logfile,'("* Starting single-ended network generation calculation..."/)')
134 write(logfile, '("- Note that this mode is EXPERIMENTAL.")')
135 Call runnetgrow2()
136
137 case ('breakdown')
138
139 write(logfile, '("* Starting single-ended breakdown product generation..."/)')
140 call runbreakdown()
141
142 case default
143 print*, '* Unknown calculation type', calctype
144 stop
145 end select
146
147
148 ! Finish things up.
149 !
150 write(logfile,'(/"================================================================")')
151 write(logfile,'(" CALCULATION COMPLETE ")')
152 write(logfile,'("================================================================"/)')
153 call flush(logfile)
154
155end Program cde
