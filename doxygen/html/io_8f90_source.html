20 !> ReadInput
21 !!
22 !! Reads the main CDE input file, and fills in the data in globaldata.f90.
23 !!
24 !! The input filename is read as the first argument from the command-line.
25 !!
26 !************************************************************************
27 !!
28 Subroutine readinput()
29 implicit none
30 integer :: narg, line, pos, ios, i,j,k, imin,imax,id
31 integer :: i1,i2,ic
32 character (len=2) :: ctemp, a1,a2
33 character (len=8) :: c_date
34 character (len=10):: c_time
35 character(len=100) :: buffer, label, comment
36 character (len=40) :: buffer2, buffer3, buffer4, templatefile,label2
37 logical :: there
38 real(8) :: r1, sum
39
40 ! Get the current date and time.
41 !
42 call date_and_time(date=c_date, time=c_time)
43
44
45 ! Read the input filename from the command line.
46 !
47 narg = iargc()
48 if ( narg .ne. 1 ) then
49 write(6,'(/"* Program usage: cde.x [input file]"/)')
50 stop
51 endif
52 call getarg (1, inputfile )
53
54
55 ! Open the input file and read the input parameters.
56 !
57 inquire( file = inputfile, exist=there )
58 if (.not.there)stop '* Specified input file does not exist'
59 Open(10, file = inputfile, status = 'unknown')
60
61
62 ! Open a *.log file with the same root as the input file.
63 !
64 Open(logfile, file = trim( inputfile )//'.log', status = 'unknown')
65 write(logfile,'(/"************************ CDE simulation ************************"/)')
66 write(logfile,'("- Start date: ",3x,a2,"/",a2,"/",a4)')c_date(7:8),c_date(5:6),c_date(1:4)
67 write(logfile,'("- Start time: ",3x,a2,":",a2,":",a2/)')c_time(1:2),c_time(3:4),c_time(5:6)
68 write(logfile,'("- The input filename is:",1x,a/)')inputfile
69 write(logfile,'("================================================================")')
70 write(logfile,'(" INPUT FILE INTERPRETATION FOLLOWS ")')
71 write(logfile,'("================================================================"/)')
72 write(logfile,'("- The input filename is:",1x,a/)')inputfile
73
74
75 ! Set default values for everything.
76 !
77 Call setiodefaults()
78
79
80 ! Loop over lines in the file, reading keywords and arguments.
81 !
82 line = 0
83 ios = 0
84 do while (ios == 0)
85 read(10, '(A)', iostat=ios) buffer
86 if (ios == 0) then
87 line = line + 1
88
89 ! Find the first space
90 pos = scan(buffer, ' ')
91 label = buffer(1:pos)
92 buffer = buffer(pos+1:)
93
94
95 ! If pos == 1, there is a space in the first position and
96 ! we assume a blank line.
97 if (pos == 1)then
98 write(logfile,*)
99 cycle
100 endif
101
102 ! Read input based on label.
103 !
104 select case (label)
105
106 case ('# ')
107 read(buffer,*,iostat=ios) comment
108 write(logfile,'("# COMMENT:",1x,a)')buffer
109
110 !***********************************************************************
111 ! GENERAL INPUT PARAMETERS.
112 !***********************************************************************
113 !
114 case ('nimage') ! Number of images along paths.
115 read(buffer,*,iostat=ios) nimage
116 write(logfile,'("- Number of images per path: ",1x,i4)')nimage
117 if (nimage <=0 )stop '* ERROR: nimage <= 0'
118
119
120 case ('calctype')
121 ! 'optpath' for path optimization,
122 ! 'molopt' for molecular property optimization (EXPERIMENTAL),
123 ! 'pathfind' for path-finding calculation,
124 ! 'evb' for EVB-based initial MEP generation (EXPERIMENTAL).
125 read(buffer, *, iostat=ios) calctype
126 if (calctype == 'optpath') then
127 write(logfile,'("- Calculation type: PATH REFINEMENT")')
128 else if (calctype == 'interp') then
129 write(logfile, '("- Calculation type: PATH INTERPOLATION")')
130 else if (calctype == 'molopt') then
131 write(logfile,'("- Calculation type: MOLECULE PROPERTY OPTIMIZATION")')
132 else if (calctype == 'pathfind') then
133 write(logfile,'("- Calculation type: REACTION-PATH FINDER")')
134 else if (calctype == 'netgrow' .or. calctype == 'netgrow2') then
135 write(logfile,'("- Calculation type: NETWORK GENERATION")')
136 else if (calctype == 'breakdown') then
137 write(logfile, '("Calculation type: BREAKDOWN NETWORK GENERATION")')
138 else
139 stop 'ERROR: Unknown calctype in input file (options: optpath, interp, molopt, &
140 & pathfind, netgrow, breakdown)'
141 endif
142
143
144 case ('pathinit') ! Path initialization method - generally has to be 'linear', not
145 ! really used other than to give internal images position values
146 read(buffer, *, iostat=ios) pathinit
147 if (pathinit == 'linear') then
148 write(logfile,'("- Initial path initialization type: LINEAR")')
149 else
150 stop 'ERROR: Unknown PathInit in input file (options: linear)'
151 endif
152
153
154 case ('startfile') ! For some calculations, reactants and product structures are required.
155 ! startfile is the xyz file containing the reactant structure....
156 read(buffer, *, iostat=ios) startfile
157 write(logfile,'("- Start-point structure file: ",1x,a)')startfile
158
159 case ('endfile') ! ..and endfile is the xyz file containing the product structure.
160 read(buffer, *, iostat=ios) endfile
161 write(logfile,'("- End-point structure file: ",1x,a)')endfile
162
163 case ('pathfile') ! xyz file containing an entire reaction path - number of images here
164 ! must be same as nimage in input file.
165 read(buffer, *, iostat=ios) pathfile
166 write(logfile,'("- Starting path file: ",1x,a)')pathfile
167
168 case ('startfrompath')
169 read(buffer, *, iostat = ios) startfrompath
170 write(logfile,'("- Start from path?: ",1x,L)')startfrompath
171
172 case ('ranseed')
173 read(buffer,*,iostat=ios)irun
174 write(logfile,'("- Random number seed: ",1x,i4)')irun
175
176
177 !***********************************************************************
178 ! EVB CALCULATION INPUT PARAMETERS.
179 !***********************************************************************
180 !
181 case ('evbtype') ! EVB-based MEP generation type (integer)
182 read(buffer,*,iostat=ios)evbtype
183 if (evbtype == 1) then
184 write(logfile,'("- EVB type 1: HARMONIC EVB METHOD")')
185 else if (evbtype == 2) then
186 write(logfile,'("- EVB type 2: DISTANCE-BASED EVB METHOD")')
187 else if (evbtype == 3) then
188 write(logfile,'("- EVB type 3: PES-BASED EVB METHOD")')
189 else
190 stop 'ERROR: Unknown evbtype in input file '
191 endif
192
193 case ('evbiter') ! Number of iterations for optimization during EVB.
194 read(buffer,*,iostat=ios)evbiter
195 if (evbiter <= 0)stop 'ERROR: evbiter not acceptable'
196 write(logfile,'("- EVB iterations: ",1x,i7)')evbiter
197
198 case ('nevbqm') ! Number of iterations for quickmin optimization after EVB.
199 read(buffer,*,iostat=ios)nevbqm
200 if (evbiter < 0)stop 'ERROR: nevbqm not acceptable'
201 write(logfile,'("- EVB quick-min iterations: ",1x,i7)')nevbqm
202
203 case ('nevbl') ! Number of intermediate structures in EVB.
204 read(buffer,*,iostat=ios)nevbl
205 if (nevbl <= 0)stop 'ERROR: nevbl not acceptable'
206 write(logfile,'("- Number of EVB intermediates: ",1x,i7)')nevbl
207
208 case ('nevbdist') ! Number of distribution tests in EVB.
209 read(buffer,*,iostat=ios)nevbdist
210 if (nevbdist <= 0)stop 'ERROR: nevbdist not acceptable'
211 write(logfile,'("- Number of EVB intermediate tests: ",1x,i7)')nevbdist
212
213 case ('evbstep') ! EVB steepest-descent step length.
214 read(buffer,*,iostat=ios)evbstep
215 if (evbstep < 0.d0)stop 'ERROR: evbstep < 0'
216 write(logfile,'("- EVB gradient step: ",1x,f14.8)')evbstep
217
218 case ('evbmaxdl') ! Maximum target distance between adjacent
219 ! images in EVB path
220 read(buffer,*,iostat=ios)evbmaxdl
221 if (evbmaxdl <= 0.d0 )stop 'ERROR: evbmaxdl <= 0'
222 write(logfile,'("- EVB maximum image-distance: ",1x,f14.8)')evbmaxdl
223
224 case ('evbalpha1') ! \alpha_1 parameter for EVB MEP
225 read(buffer,*,iostat=ios)evbalpha1
226 if (evbalpha1 < 0.d0) stop 'ERROR: evbalpha1 < 0'
227 write(logfile,'("- EVB alpha1: ",1x,f14.8)')evbalpha1
228
229 case ('evbalpha2') ! \alpha_2 parameter for EVB MEP
230 read(buffer,*,iostat=ios)evbalpha2
231 if (evbalpha1 < 0.d0) stop 'ERROR: evbalpha2 < 0'
232 write(logfile,'("- EVB alpha2: ",1x,f14.8)')evbalpha2
233
234 case ('evbvrep') ! Logical flag indicating whether to use repulsive
235 ! potential during EVB calculation.
236 read(buffer,*,iostat=ios)evbvrep
237 if (evbvrep) then
238 write(logfile,'("- Use Vrep in EVB: TRUE")')
239 else
240 write(logfile,'("- Use Vrep in EVB: FALSE")')
241 endif
242
243 case ('alphavbe') ! Strength parameter for adding-in approximate additive
244 ! bond-strength changes during double-ended pathfinding.
245 read(buffer,*,iostat=ios)alphavbe
246 write(logfile,'("- Strength parameter for additive bonds in &
247 double-ended GDS: ",1x,f14.8)')alphavbe

249
250 !***********************************************************************
251 ! MEP CALCULATION INPUT PARAMETERS.
252 !***********************************************************************
253 !
254 case ('pathoptmethod') ! MEP refinement method - 'cineb' for (climbing image) nudged

258 write(logfile,'("- Path refinement method: CLIMBING-IMAGE NUDGED ELASTIC BAND")')
259 else if (pathoptmethod == 'zts') then
260 write(logfile,'("- Path refinement method: ZERO-TEMPERATURE STRING")')

263 endif
264

267 ! steepest - steepest descents method

270 ! dfp - BFGS-type scheme, currently unstable!
271 ! bmk - Runge-Kutta method
272 ! hball - Heavy rolling ball method.
273 read(buffer, *, iostat=ios) nebmethod
274 if (nebmethod == 'pathquad') then
275 write(logfile,'("- (CI)NEB optimization method: QUADRATIC INTERPOLATION")')
276 else if (nebmethod == 'steepest') then
277 write(logfile,'("- (CI)NEB optimization method: STEEPEST DESCENT")')
278 else if (nebmethod == 'quickmin') then
279 write(logfile,'("- (CI)NEB optimization method: QUICKMIN")')
280 else if (nebmethod == 'fire') then
281 write(logfile,'("- (CI)NEB optimization method: FIRE")')
282 else if (nebmethod == 'dfp') then
283 write(logfile,'("- (CI)NEB optimization method: DFP (BFGS)")')
284 else if (nebmethod == 'bmk') then
285 write(logfile,'("- (CI)NEB optimization method: BMk (Runge-Kutta)")')
286 else if (nebmethod == 'hball') then
287 write(logfile,'("- (CI)NEB optimization method: HEAVY-BALL ")')
288 else
289 stop 'ERROR: UNKNown NEBmethod in input file (options: pathquad, steepest, &
290 & quickmin, fire, dfp, bmk, hball)'
291 endif
292
293 case ('nebconv') ! NEB convergence criteria on RMS forces, in au.
294 read(buffer,*,iostat=ios) nebconv
295 if (nebconv <= 0.d0) then
296 stop '* ERROR: NEBconv <= 0.0'
297 else
298 write(logfile,'("- NEB NORM force-convergence criterion (au): ",1x,f14.8)')nebconv
299 endif
300
301 case ('nebmaxconv') ! Convergence criterion on maximum force during NEB, in au.
302 read(buffer,*,iostat=ios) nebmaxconv
303 if (nebmaxconv <= 0.d0) then
304 stop '* ERROR: NEBmaxconv <= 0.0'
305 else
306 write(logfile,'("- NEB MAXIMUM force-convergence criterion (au): ",1x,f14.8)')nebmaxconv
307 endif
308
309 case ('cithresh') ! Threshold on RMS forces at which climbing-image variant is engaged.
310 read(buffer,*,iostat=ios) cithresh
311 if (cithresh <= 0.d0) then
312 stop '* ERROR: CIthresh <= 0.0'
313 else
314 write(logfile,'("- Climbing-image start criterion (au): ",1x,f14.8)')cithresh
315 endif
316
317 case ('stripinactive') ! Remove inactive molecules from path.
318 read(buffer, *, iostat = ios) stripinactive
319 write(logfile,'("- Strip inactive molecules before path optimization?: ",1x,L)')stripinactive
320
321 case ('alignedatoms')
322 read(10, *, iostat = ios) atomidx(1),atomidx(2),atomidx(3)
323 write(logfile,'("- Aligned atoms for NEB refinements: ",1x,3i5)')atomidx(1:3)
324
325 case ('nebiter')
326 read(buffer,*,iostat=ios)nebiter
327 write(logfile,'("- Maximum number of NEB iteractions: ",1x,i6)')nebiter
328
329 case ('neboutfreq')
330 read(buffer,*,iostat=ios)neboutfreq
331 write(logfile,'("- NEB output frequency: ",1x,i6)')neboutfreq
332
333 case ('nebspring')
334 read(buffer,*,iostat=ios)nebspring
335 write(logfile,'("- NEB spring constant: ",1x,f14.6)')nebspring
336
337 case ('nebstep')
338 read(buffer,*,iostat=ios)nebstep
339 write(logfile,'("- NEB optimization step-size: ",1x,f14.6)')nebstep
340
341 case ('optendsbefore')
342 read(buffer,*,iostat = ios)optendsbefore
343 write(logfile,'("- Optimize end-points before NEB run: ",1x,L)')optendsbefore
344
345 case ('reconnect')
346 read(buffer,*,iostat=ios)reconnect
347 write(logfile,'("- Reconnect end-points after optimization: ",1x,L)')reconnect
348
349 case ('optendsduring')
350 read(buffer,*,iostat = ios)optendsduring
351 write(logfile,'("- Optimize end-points during NEB run: ",1x,L)')optendsduring
352
353 case ('idppguess')
354 read(buffer, *, iostat=ios) idpppath
355 write(logfile,'("- IDPP interpolation : ",1x,L)')idpppath
356
357
358 case ('projforcetype')
359 read(buffer, *, iostat=ios) projforcetype
360 write(logfile,'("- Using NEB projected forces of type: ",1x,I2)')projforcetype
361
362 case ('nebrestrend')
363 read(buffer,*,iostat = ios)nebrestrend
364 write(logfile,'("- Use graph-restrain potential during NEB: ",1x,L)')nebrestrend
365
366 case ('vsthresh')
367 read(buffer, *, iostat = ios) vsthresh
368 write(logfile,'("- Make variable spring strengths with threshold?: ",1x,f8.6)')vsthresh
369
370
371 !***********************************************************************
372 ! GDS PARAMETERS.
373 !***********************************************************************
374 !
375 case('reactiveatomtypes{')
376 nreactivetypes = 0
377 do i = 1, ntypemax
378 read(10,*,iostat=ios) ctemp
379 if (ctemp /= '}') then
380 nreactivetypes = nreactivetypes + 1
381 reactivetype(nreactivetypes) = ctemp
382 else if (ctemp == '}') then
383 EXIT
384 endif
385 enddo
386 write(logfile,'("- Number of reactive atom types: ",1x,i4)')nreactivetypes
387 do i = 1, nreactivetypes
388 write(logfile,'("- Reactive atom labels: ",1x,A)')reactivetype(i)
389 enddo
390
391 case('valencerange{')
392 nvalcon = 0
393 do i = 1, ntypemax
394 read(10,'(A)',iostat=ios)buffer2
395 if (buffer2(1:1) /= '}') then
396 a2 = ''
397 if (index(buffer2,'fz').ne. 0 ) &
398 read(buffer2,*)a1,imin,imax, a2
399 if (index(buffer2,'fz').eq. 0 ) &
400 read(buffer2,*)a1,imin,imax
401 nvalcon = nvalcon + 1
402 valatom(nvalcon) = a1
403 valrange(nvalcon,1) = imin
404 valrange(nvalcon,2) = imax
405 if (trim(a2) .eq. 'fz' ) valfz(nvalcon) = .true.
406 else if (buffer2(1:1) == '}') then
407 EXIT
408 endif
409 enddo
410 write(logfile,'("- Number of valence range constraints: ",1x,i4)')nvalcon
411 do i = 1, nvalcon
412 write(logfile,'("- Valence range constraint: ",1x,i3,1x,A,1x,i4,1x,i4)')i,valatom(i),valrange(i,1),valrange(i,2)
413 enddo
414
415 case('reactivevalence{')
416 nrxval = 0
417 do i = 1, ntypemax
418 read(10,'(A)',iostat=ios)buffer2
419 if (buffer2(1:1) /= '}') then
420 read(buffer2,*)a1,a2,imin,imax
421 nrxval = nrxval + 1
422 rxvalatom(nrxval,1) = a1
423 rxvalatom(nrxval,2) = a2
424 rxvalrange(nrxval,1) = imin
425 rxvalrange(nrxval,2) = imax
426 else if (buffer2(1:1) == '}') then
427 EXIT
428 endif
429 enddo
430 write(logfile,'("- Number of reactive valence constraints: ",1x,i4)')nrxval
431 do i = 1, nrxval
432 write(logfile,'("- Reactive valence range constraint: ",1x,i3,1x,A,1x,A,1x,i4,1x,i4)') &
433 i,rxvalatom(i,1),rxvalatom(i,2),rxvalrange(i,1),rxvalrange(i,2)
434 enddo
435
436 case('reactiveatoms{')
437 reactive(:) = .false.
438 do i = 1, namax
439 read(10,'(A)',iostat=ios) buffer2
440 pos = scan(buffer2, ' ')
441 label2 = buffer2(1:pos)
442 buffer2 = buffer2(pos+1:)
443 select case(label2)
444 case('range')
445 read(buffer2,*,iostat=ios)imin,imax
446 write(logfile,'("- Reactive atom range: ",1x,i4," - ",i4)')imin,imax
447 do j = imin, imax
448 reactive(j) = .true.
449 enddo
450 case('id')
451 read(buffer2,*,iostat=ios)id
452 reactive(id) = .true.
453 case('all')
454 if (.not. startfrompath) &
455 open(12, file = startfile, status = 'unknown')
456 if ( startfrompath) &
457 open(12, file = pathfile, status = 'unknown')
458 read(12, *, iostat=ios) imax
459 do j = 1, imax
460 reactive(j) = .true.
461 enddo
462 close(12)
463 case('}')
464 EXIT
465 case default
466 print*,'* Error in reactiveatoms{} block'
467 stop
468 end select
469 enddo
470
471 case('essentialmoveatoms{')
472 essentialmoves(:) = .false.
473 do i = 1, namax
474 read(10,'(A)',iostat=ios) buffer2
475 pos = scan(buffer2, ' ')
476 label2 = buffer2(1:pos)
477 buffer2 = buffer2(pos+1:)
478 select case(label2)
479 case('range')
480 read(buffer2,*,iostat=ios)imin,imax
481 write(logfile,'("- Essential atom range: ",1x,i4," - ",i4)')imin,imax
482 do j = imin, imax
483 essentialmoves(j) = .true.
484 enddo
485 case('id')
486 read(buffer2,*,iostat=ios)id
487 essentialmoves(id) = .true.
488 case('all')
489 open(12, file = startfile, status = 'unknown')
490 read(12, *, iostat=ios) imax
491 do j = 1, imax
492 essentialmoves(j) = .true.
493 enddo
494 close(12)
495 case('}')
496 EXIT
497 case default
498 print*,'* Error in essentialmoveatoms{} block'
499 stop
500 end select
501 enddo
502 ! Count number of essential moves...
503 !
504 nessentialatoms = 0
505 do i = 1, namax
506 if (essentialmoves(i))nessentialatoms = nessentialatoms + 1
507 enddo
508
509 case('essentialatomsinmols{')
510 essentialatomsinmols(:) = 0
511 ic = 0
512 do i = 1, namax
513 read(10,'(A)',iostat=ios) buffer2
514 pos = scan(buffer2, ' ')
515 label2 = buffer2(1:pos)
516 buffer2 = buffer2(pos+1:)
517 select case(label2)
518 case('range')
519 read(buffer2,*,iostat=ios)imin,imax
520 write(logfile,'("- Essential atoms in mols range: ",1x,i4," - ",i4)')imin,imax
521 do j = imin, imax
522 ic = ic + 1
523 essentialatomsinmols(ic) = j
524 enddo
525 case('id')
526 read(buffer2,*,iostat=ios)id
527 ic = ic + 1
528 essentialatomsinmols(ic) = id
529 case('}')
530 EXIT
531 case default
532 print*,'* Error in essentialatoms{} block'
533 stop
534 end select
535 enddo
536
537 ! Count number of essential moves...
538 !
539 nessentialatomsinmols = ic
540
541 case('essentialatoms{')
542 essential(:) = .false.
543 do i = 1, namax
544 read(10,'(A)',iostat=ios) buffer2
545 pos = scan(buffer2, ' ')
546 label2 = buffer2(1:pos)
547 buffer2 = buffer2(pos+1:)
548 select case(label2)
549 case('range')
550 read(buffer2,*,iostat=ios)imin,imax
551 write(logfile,'("- Essential atom range: ",1x,i4," - ",i4)')imin,imax
552 do j = imin, imax
553 essential(j) = .true.
554 enddo
555 case('id')
556 read(buffer2,*,iostat=ios)id
557 essential(id) = .true.
558 case('all')
559 open(12, file = startfile, status = 'unknown')
560 read(12, *, iostat=ios) imax
561 do j = 1, imax
562 essential(j) = .true.
563 enddo
564 close(12)
565 case('}')
566 EXIT
567 case default
568 print*,'* Error in essentialatoms{} block'
569 stop
570 end select
571 enddo
572
573 case('allowedbonds{')
574 nallowbonds = 0
575 do i = 1, ntypemax
576 read(10,'(A)',iostat=ios)buffer2
577 if (buffer2(1:1) /= '}') then
578 read(buffer2,*)a1,a2,imin
579 nallowbonds = nallowbonds + 1
580 allowbondsatom(nallowbonds,1) = a1
581 allowbondsatom(nallowbonds,2) = a2
582 allowbondsmax(nallowbonds) = imin
583 else if (buffer2(1:1) == '}') then
584 EXIT

586 enddo

593 case('fixedbonds{')

622
623 case('nbstrength')

844 !! Reads the set of graph moves used during GDS simulation from

849 !******************************************************************

851 Subroutine readgraphmoves( movefile )
852 implicit none
853 character(len=50) :: movefile
854 logical :: there
855 integer :: line, ios, pos, i, pos2
856 character(len=100) :: buffer, label, comment, buffer2, label2, gwritefmt
857 real(8) :: sum

862 ! Check and open movefile.
863 !
864 inquire( file = movefile, exist=there )
865 if (.not.there)stop '* Specified move file does not exist'
866 Open(20, file = movefile, status = 'unknown')
867 write(logfile,'("================================================================")')
868 write(logfile,'(" MOVE FILE INTERPRETATION FOLLOWS... ")')

872 ! Loop over lines in the file, reading keywords and arguments.
873 !
874 ngmove = 0
875 line = 0
876 ios = 0
877 do while (ios == 0)
878 read(20, '(A)', iostat=ios) buffer
879 if (ios == 0) then
880 line = line + 1
881
882 ! Find the first space
883 pos = scan(buffer, ' ')
884 label = buffer(1:pos)
885 buffer = buffer(pos+1:)
886
887 ! If pos == 1, there is a space in the first position and we assume a blank line.
888 if (pos == 1) then
889 ! write(logfile,*)label,buffer
890 cycle

893 ! Read input based on label.
894 !
895 select case (label)
896
897 ! Comment line
898 case ('# ')
899 read(buffer, *, iostat=ios) comment

901 write(logfile, '(A, " ", A, /)') trim(label), trim(buffer)
902

904 case ('move')
905 ngmove = ngmove + 1
906 write(logfile, '("*** Graph move number:", 1x, i4, " ***"/)') ngmove
907

910 pos2 = scan(buffer2 , ' ')
911 label2 = buffer2(1:pos2)
912 buffer2 = buffer2(pos2+1:)
913 read(buffer2, *, iostat=ios) namove(ngmove)
914 write(logfile, '("- Number of atoms:", 1x, i4)') namove(ngmove)
915
916 ! Is this a bond-changing move....?
917 if (namove(ngmove) > 0) then
918 write(logfile, '("- Bond-changing move detected ")')
919 write(gwritefmt, '("(", I1, "I3)")') namove(ngmove)
920
921 ! Next, read the starting and ending graph - the first line is a '-', followed
922 ! by the namove x namove matrix on the next namove lines.
923 write(logfile, '("- Start graph: ")')
924 read(20, '(A)', iostat=ios) buffer2
925 do i = 1, namove(ngmove)
926 read(20, *, iostat=ios) gmstart(ngmove, i, 1:namove(ngmove))
927 write(logfile, gwritefmt) gmstart(ngmove, i, 1:namove(ngmove))
928 enddo
929 read(20, '(A)', iostat=ios) buffer2
930 write(logfile, '("- End graph: ")')
931 do i = 1, namove(ngmove)
932 read(20, *, iostat=ios) gmend(ngmove, i, 1:namove(ngmove))
933 write(logfile, gwritefmt) gmend(ngmove, i, 1:namove(ngmove))
934 enddo
935 read(20, '(A)', iostat=ios) buffer2
936
937 ! Now read the labels of the atoms, if any.
938 read(20, '(A)', iostat=ios) buffer2
939 pos2 = scan(buffer2, ' ')
940 label2 = buffer2(1:pos2)
941 buffer2 = buffer2(pos2+1:)
942 read(buffer2, *, iostat=ios) (movelabel(ngmove, i), i=1, namove(ngmove))
943 write(logfile, '("- Allowed atom labels: ")')
944 write(gwritefmt, '("(", I1, "A2)")') namove(ngmove)
945 write(logfile, gwritefmt) movelabel(ngmove, 1:namove(ngmove))
946
947 ! Now read the move probability.
948 read(20, '(A)', iostat=ios) buffer2
949 pos2 = scan(buffer2, ' ')
950 label2 = buffer2(1:pos2)
951 buffer2 = buffer2(pos2+1:)
952 read(buffer2, *, iostat = ios) moveprob(ngmove)
953 write(logfile, '("Move probability: ", F5.3/)') moveprob(ngmove)
954 call flush(logfile)
955
956 ! ..or is it a molecular charge-changing move?
957 else if (namove(ngmove) == 0) then
958
959 write(logfile, '("- Molecular charge-changing move detected ")')
960 read(20, '(A)', iostat=ios) buffer2 ! Reading the '-'...
961 read(20, '(A)', iostat=ios) buffer2
962 pos2 = scan(buffer2, ' ')
963 label2 = buffer2(1:pos2)
964 movetype(ngmove) = label2
965 read(20, '(A)', iostat=ios) buffer2 ! Reading the '-'...
966
967 ! Now read the move probability.
968 read(20, '(A)', iostat=ios) buffer2
969 pos2 = scan(buffer2, ' ')
970 label2 = buffer2(1:pos2)
971 buffer2 = buffer2(pos2+1:)
972 read(buffer2, *, iostat = ios) moveprob(ngmove)
973
974 endif
975
976 end select
977
978 endif
979 enddo
980
981 ! Make sure to normalize probabilities of graph moves.
982 sum = 0.d0
983 do i = 1, ngmove
984 sum = sum + moveprob(i)
985 enddo
986 do i = 1, ngmove
987 moveprob(i) = moveprob(i) / sum
988 enddo
989 write(logfile,'("================================================================")')
990 write(logfile,'(" MOVE FILE INTERPRETATION COMPLETE ")')
991 write(logfile,'("================================================================")')
992
993 write(logfile,'(/"* Finished reading graph moves file..."/)')
994
995 return
996 end Subroutine readgraphmoves
