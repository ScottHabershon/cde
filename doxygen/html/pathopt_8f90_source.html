21 !************************************************************************
22 !> InterpolatePath
23 !!
24 !! Interpolates a reaction path, without optimisation.
25 !!
26 !! - rp: Generated interpolation of reaction path.
27 !!
28 !************************************************************************
29 !
30 subroutine interpolatepath(rp)
31 type(rxp), intent(out) :: rp
32
33 integer :: nmol(2)
34 logical :: ldum
35
36 ! Initialize the reaction path object.
37 Call newpath(rp, startfrompath, startfile, endfile, pathfile, nimage, &
38 pathinit, .false., idum)
39
40 ! Print the initial path.
41 Call printpathtofile(rp, 'linear_path.xyz')
42 write(logfile,'("- Initial path written to linear_path.xyz")')
43
44 ! If stripinactive is true, then we strip out the inactive molecules (i.e. those not
45 ! involved in the reaction).
46 if (stripinactive) then
47 write(logfile, '("*** Stripping inactive molecules from path...")')
48 Call stripinactivefrompath(rp, 'linear_path_stripped.xyz', fixeddof, fixedatom, ndofconstr, natomconstr, .true.)
49 write(logfile, '("- Stripped initial path written to linear_path_stripped.xyz")')
50
51 ! Now delete the original path and read the new one.
52 Call deletepath(rp)
53 ! ..and create a new path by reading from file...
54 Call newpath(rp, .true., startfile, endfile, 'linear_path_stripped.xyz', nimage, &
55 pathinit, .false., idum)
56 endif
57
58 if (idpppath) then
59 ! Initialize constraints in the path.
60 Call setpathconstraints(rp, ndofconstr, fixeddof, natomconstr, fixedatom)
61
62 ! Determine initial graphs and molecules for end-points.
63 call getgraph(rp%cx(1))
64 call getmols(rp%cx(1))
65 nmol(1) = rp%cx(1)%nmol
66 call getgraph(rp%cx(rp%nimage))
67 call getmols(rp%cx(rp%nimage))
68 nmol(2) = rp%cx(rp%nimage)%nmol
69
70 ! Do we optimize the end-points before NEB?
71 if (optendsbefore) then
72 call abinitio(rp%cx(1), 'optg', ldum)
73 call abinitio(rp%cx(rp%nimage), 'optg', ldum)
74 call getgraph(rp%cx(1))
75 call getmols(rp%cx(1))
76 call getgraph(rp%cx(rp%nimage))
77 call getmols(rp%cx(rp%nimage))
78 if (nmol(1) .ne. rp%cx(1)%nmol .or. nmol(2) .ne. rp%cx(rp%nimage)%nmol ) then
79 write(6,'("Warning: Number of molecules changed during optimization!: was: ",2I2,", is: ", 2I2)') &
80 nmol(1:2), rp%cx(1)%nmol , rp%cx(rp%nimage)%nmol
81 write(logfile,'("Warning: Number of molecules changed during optimization!: was: ",2I2,", is: ", 2I2)') &
82 nmol(1:2), rp%cx(1)%nmol , rp%cx(rp%nimage)%nmol
83 endif
84 endif
85
86 ! If required, redraw the internal images.
87 if (reconnect) then
88 rp%coeff(:,:,:) = 0.d0
89 call fouriertopath(rp)
90 endif
91
92 ! Recalculate graphs and mols.
93 call getgraph(rp%cx(1))
94 call getmols(rp%cx(1))
95 call getgraph(rp%cx(rp%nimage))
96 call getmols(rp%cx(rp%nimage))
97
98 ! Calculate IDPP path.
99 call findidpppath(rp, nebiter, nebconv, nebstep, nebspring)
100
101 ! Print the IDPP path.
102 Call printpathtofile(rp,'idpp_path.xyz')
103 write(logfile,'("- IDPP-refined path written to idpp_path.xyz")')
104 endif
105
106 return
107 end subroutine interpolatepath

192 call getmols(rp%cx(1))
193 nmol(1) = rp%cx(1)%nmol
194 call getgraph(rp%cx(rp%nimage))
195 call getmols(rp%cx(rp%nimage))

202 call abinitio(rp%cx(1), 'optg', ldum)
203 call abinitio(rp%cx(rp%nimage), 'optg', ldum)
204 call getgraph(rp%cx(1))
205 call getmols(rp%cx(1))
206 call getgraph(rp%cx(rp%nimage))
207 call getmols(rp%cx(rp%nimage))
208 if (nmol(1) .ne. rp%cx(1)%nmol .or. nmol(2) .ne. rp%cx(rp%nimage)%nmol ) then
209 write(6,'("Warning: Number of molecules changed during optimization!: was: ",2I2,", is: ", 2I2)') &
210 nmol(1:2), rp%cx(1)%nmol , rp%cx(rp%nimage)%nmol
211 write(logfile,'("Warning: Number of molecules changed during optimization!: was: ",2I2,", is: ", 2I2)') &
212 nmol(1:2), rp%cx(1)%nmol , rp%cx(rp%nimage)%nmol

214 endif
215
216 ! If required, redraw the internal images.
217 if (reconnect) then
218 rp%coeff(:,:,:) = 0.d0

224 variablespring = .false.

229 call getmols(rp%cx(1))

238 call findidpppath(rp, nebiter*250, nebconv*0.1d0, nebstep, nebspring)

241 ! Sort out orientation of images.
242 !
243 ! Call SortOrientation(rp)
244

246 !

252 n1 = 1
253 n2 = rp%nimage

279 ! Add restraint forces to end-points if required.

281 if (nebrestrend .and. optendsduring .and. (.not. optendsbefore) ) then

298 open(14, file = trim( inputfile )//'.energy-neb-start', status = 'unknown')

304 write(14, '(3(1x,f14.8))') lambda, rp%cx(i)%vcalc, (rp%cx(i)%vcalc - rp%cx(1)%vcalc)

314 !

316 call printpathtofile(rp, 'full_neb_traj.xyz')

324 case(2)

327 call getprojforces3(rp, .false., .false., optendsduring)

329 call getforcenorm(rp, fnorm1, fmax, n1, n2)

331 case(1)
332 call getprojforces1(rp, .true., .false., optendsduring)
333 case(2)
334 call getprojforces2(rp, .true., .false., optendsduring)
335 case(3)
336 call getprojforces3(rp, .true., .false., optendsduring)
337 end select
338 call getforcenorm(rp, fnorm2, fmax, n1, n2)
339 write(18,*)' 0 ', fnorm2, fmax
340 write(logfile, '("*** Initial force norms: ")')
341 write(logfile, '("* |Forces| WITHOUT springs = ",1x,f14.8," au ")') fnorm1
342 write(logfile, '("* |Forces| WITH springs = ",1x,f14.8," au ")') fnorm2
343 write(logfile, '("* Maximum force WITH springs = ",1x,f14.8," au "/)') fmax
344
345 ! Now perform optimization of the path based on the method in NEBmethod:
346 !
347 select case (nebmethod)
348 case('quickmin') !< Quickmin, with Euler update.
349 write(logfile, '("* Running QUICKMIN optimization..."/)')
350
351 ! Zero initial momenta.
352 do i = n1, n2
353 rp%cx(i)%p = 0.d0
354 enddo
355
356 case('steepest') !< Simple steepest descent, taking NEBstep * Force step each iteration
357 write(logfile, '("* Running STEEPEST DESCENT optimization..."/)')
358
359 case('fire')
360 write(logfile, '("* Running FIRE optimization..."/)')
361 allocate(v(rp%nimage, rp%na*3), f(rp%nimage, rp%na*3), alph(rp%nimage))
362 nmin = 5
363 f_inc = 1.10d0 ; f_dec = 0.50d0 ; f_alp = 0.990d0 ; alph_start = 0.10d0
364 delt = nebstep/10.0 ; alph = alph_start ; nmin_count = 0
365 do i = n1, n2
366 rp%cx(i)%p = 0.0d0
367 do j = 1, rp%na
368 rp%cx(i)%p(1:3, j) = rp%cx(i)%p(1:3, j) + delt * rp%cx(i)%force(1:3, j)/2.0d0

393 idof = 0
394 do j = 1, rp%na

397 idof = idof + 1
398 if (.not. rp%cx(i)%FixedDOF(idof)) then

406 endif
407 enddo
408 enddo

411 do i = n1, n2
412 idof = 0
413 do j = 1, rp%na
414 if (.not. rp%cx(i)%fixedatom(j)) then
415 do k = 1, 3
416 idof = idof + 1
417 if (.not. rp%cx(i)%FixedDOF(idof)) then
418 rp%cx(i)%r(k, j) = rp%cx(i)%r(k, j) + nebstep * rp%cx(i)%force(k, j)

420 enddo
421 else
422 idof = idof + 3
423 endif
424 enddo
425 enddo

428 do i = n1, n2

431 if (norm2(f(i, :)) .gt. 0.00001) then
432 v(i, :) = (1.0d0-alph(i)) * v(i, :) + alph(i) * f(i, :) * norm2(v(i, :))/norm2(f(i, :))

434 if (pfire > 0.0d0 .and. nmin_count > nmin) then
435 delt = min(delt * f_inc, 1.0 * nebstep)
436 alph(i) = alph(i) * f_alp
437 elseif (pfire <= 0.0d0) then
438 nmin_count = 0
439 delt = delt * f_dec
440 v(i, :) = 0.0d0
441 rp%cx(i)%p = 0.0d0
442 alph(i) = alph_start
443 endif
444 if (pfire > 0.0d0) nmin_count = nmin_count + 1
445 do j = 1, rp%na
446 rp%cx(i)%r(1:3, j) = rp%cx(i)%r(1:3, j) + delt * v(i, (j-1)*3+1:(j-1)*3+3)
447 ! rp%cx(i)%p(1:3,j) = v(i,(j-1)*3+1:(j-1)*3+3)*rp%cx(i)%mass(j) + delt * rp%cx(i)%force(1:3,j)
448 !v(i,(j-1)*3+1:(j-1)*3+3) = rp%cx(i)%p(1:3,j) / rp%cx(i)%mass(j)
449 rp%cx(i)%p(1:3, j) = v(i, (j-1)*3+1:(j-1)*3+3) + delt * rp%cx(i)%force(1:3, j)
450 v(i, (j-1)*3+1:(j-1)*3+3) = rp%cx(i)%p(1:3, j)
451 enddo
452 enddo
453
454 case default
455
456 end select
457
458 ! Recalculate energy and projected forces.
459 !
460 call getpathgradients(rp, success, .false.)
461
462 ! Remove overall translation and rotation.
463 ! TBD
464
465 ! Add restraint forces to end-points if required.
466 !
467 if (nebrestrend .and. optendsduring .and. (.not. optendsbefore)) then
468 gg = 0.50d0*(1.0d0+tanh((fnorm1-xe/2.0d0)*aa))
469 call graphconstraints(rp%cx(1), gdsrestspring*gg, nbstrength*gg, nbrange, kradius)
470 call graphconstraints(rp%cx(rp%nimage), gdsrestspring*gg, nbstrength*gg, nbrange, kradius)
471 rp%cx(1)%vcon = 0.0d0 ; rp%cx(rp%nimage)%vcon = 0.0d0
472 endif
473
474 ! Check convergence - calculate perpendicular forces without spring.
475 !
476 select case (projforcetype)
477 case(1)
478 call getprojforces1(rp, .false., ci_flag, optendsduring)
479 case(2)
480 call getprojforces2(rp, .false., ci_flag, optendsduring)
481 case(3)
482 call getprojforces3(rp, .false., ci_flag, optendsduring)
483 case default
484 end select
485 call getforcenorm(rp, fnorm1, fmax, n1, n2)
486 select case (projforcetype)
487 case(1)
488 call getprojforces1(rp, .true., ci_flag, optendsduring)
489 case(2)
490 call getprojforces2(rp, .true., ci_flag, optendsduring)
491 case(3)
492 call getprojforces3(rp, .true., ci_flag, optendsduring)
493 case default
494 end select
495 call getforcenorm(rp, fnorm2, fmax, n1, n2)
496 write(18, *) iter, fnorm2, fmax
497 write(logfile, '(/"*** Optimization progress summary at iteration: ",1x,i5)') iter
498 write(logfile, '("* |Forces| WITHOUT springs = ",1x,f14.8," au ")') fnorm1
499 write(logfile, '("* |Forces| WITH springs = ",1x,f14.8," au ")') fnorm2
500 write(logfile, '("* Maximum force WITH springs = ",1x,f14.8," au "/)') fmax
501
502 ! Determine variable springs, if being used:
503 !
504 if (variablespring) then
505 call variablesprings(rp)
506 endif
507
508 ! Do we turn on CINEB forces?
509 !
510 if (fnorm2 <= cithresh) then
511 write(logfile, '("* |Forces| < CIthresh :: TURNING ON CLIMBING-IMAGE")')
512 ci_flag = .true.
513 endif
514
515 ! Do we turn on variable spring strength?
516 !
517 if (fnorm2 <= vsthresh .and. vsthresh /= 0.0d0 ) then
518 write(logfile, '("* |Forces| < VSthresh :: TURNING ON VARIABLE SPRING")')
519 variablespring = .true.
520 endif
521
522 ! Output data every NEBoutfreq steps.
523 !
524 if (mod(iter,neboutfreq) == 0) then
525 lambda = 0.0
526 do i = 1, rp%nimage
527 write(17, '(3(1x,f14.8))') lambda, rp%cx(i)%vcalc, (rp%cx(i)%vcalc - rp%cx(1)%vcalc)
528 if (i .lt. rp%nimage) lambda = lambda + norm2(reshape(rp%cx(i+1)%r-rp%cx(i)%r, (/rp%na*3/)))
529 enddo
530 write(17, *)
531 write (x1,fmt4) iter
532 ! If writing every step, just append to a centralised trajectory instead of multiple files.
533 if (neboutfreq == 1) then
534 call printpathtofile(rp, 'full_neb_traj.xyz', .true.)
535 else
536 call printpathtofile(rp, trim( inputfile ) //'_'//trim(x1)//'.xyz')
537 endif
538 write(logfile, '(/"*** Current path output: ITERATION =",1x,i5,1x,":: OUTPUT FILE =",1x,A/)') iter, &
539 trim(inputfile) //'_'//trim(x1)//'.xyz'
540 endif
541
542 ! Are we converged?
543 !
544! if (fnorm1 <= NEBconv .and. fmax <= NEBmaxconv ) then ! changed to fnorm 2 !
545 if (fnorm2 <= nebconv) then ! changed to fnorm 2 !
546 write(logfile, '("* |Forces| < NEBconv :: CONVERGED")')
547 exit
548 endif
549
550 ! New convergence, assuming CI-NEB.
551 !
552 ! Emax = -1d6
553 ! do i = n1, n2
554 ! if (rp%cx(i)%vcalc > Emax ) then
555 ! Emax = rp%cx(i)%vcalc
556 ! imax = i
557 ! endif
558 ! enddo
559 ! fnorm1 = 0.d0
560 ! isum = 0
561 ! isum2 = 0
562 ! fnorm2 = 0.d0
563 ! do i = n1, n2
